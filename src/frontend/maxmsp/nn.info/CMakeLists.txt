cmake_minimum_required(VERSION 3.10 FATAL_ERROR)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(C74_MIN_API_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../min-api)
include(${C74_MIN_API_DIR}/script/min-pretarget.cmake)
set("${PROJECT_NAME}_EXTERN_OUTPUT_NAME" "netBender.info" CACHE STRING "External output name" FORCE)

if (APPLE)
	set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0")
endif()


#############################################################
# MAX EXTERNAL
#############################################################

if (NOT VERSION)
	if (CMAKE_PROJECT_VERSION)
		set(VERSION "v${CMAKE_PROJECT_VERSION}")
	else()
		set(VERSION "v0.1.0")
	endif()
endif()
message(STATUS "Building netBender.info version: ${VERSION}")
add_definitions(-DVERSION="${VERSION}")



set(
SOURCE_FILES
	nn.info.cpp
)

add_library( 
	${PROJECT_NAME} 
	MODULE
	${SOURCE_FILES}
)

include_directories( 
	"${C74_INCLUDES}"
	"${CMAKE_CURRENT_SOURCE_DIR}/../shared"
	"${CMAKE_CURRENT_SOURCE_DIR}/../../shared"
)

if (MSVC)
	include_directories(${VCPKG_INCLUDE_DIR})
	link_directories(${VCPKG_LIB_DIR})
endif()


target_link_libraries(${PROJECT_NAME} PRIVATE backend)

if (NOT CURL_LIBRARY)
	if (UNIX)
		set(CONDA_ENV_PATH "${CMAKE_SOURCE_DIR}/../env")
		set(CURL_INCLUDE_DIR "${CONDA_ENV_PATH}/include")
		set(CURL_LIBRARY "${CONDA_ENV_PATH}/lib/libcurl.dylib")
	elseif(MSVC)
		set(VCPKG_PATH "${CMAKE_SOURCE_DIR}/../vcpkg")
		set(CURL_INCLUDE_DIR "${VCPKG_PATH}/packages/curl_x64-windows/include")
		set(CURL_LIBRARY "${VCPKG_PATH}/packages/curl_x64-windows/lib/libcurl.lib")
	endif()
endif()
include_directories(${CURL_INCLUDE_DIR})

target_link_libraries(${PROJECT_NAME} PRIVATE nlohmann_json::nlohmann_json)
target_link_libraries(${PROJECT_NAME} PRIVATE ${CURL_LIBRARY})



include(${C74_MIN_API_DIR}/script/min-posttarget.cmake)

# min-posttarget.cmake resets CXX_STANDARD to 17; override back to 20
# (nn.info.cpp uses C++20 features like std::unordered_map::contains)
set_property(TARGET ${PROJECT_NAME} PROPERTY CXX_STANDARD 20)
set_property(TARGET ${PROJECT_NAME} PROPERTY CXX_STANDARD_REQUIRED ON)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_20)




if (APPLE)
	set_target_properties(${PROJECT_NAME} PROPERTIES
		BUILD_WITH_INSTALL_RPATH TRUE
		INSTALL_RPATH "${TORCH_INSTALL_PREFIX}/lib;${torch_dir}/libtorch/lib;@loader_path/../../support;@loader_path/../../../support;/Library/Application Support/NetBender/lib;${CONDA_ENV_PATH}/lib"
	)

	if (BUNDLE_DEPENDENCIES)
		if (EXISTS "${CMAKE_SOURCE_DIR}/../env/ssl/cert.pem")
			add_custom_command(
				TARGET ${PROJECT_NAME}
				POST_BUILD 
				COMMAND ${CMAKE_COMMAND} -E copy  "${CMAKE_SOURCE_DIR}/../env/ssl/cert.pem" "$<TARGET_FILE_DIR:${PROJECT_NAME}>"
				COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../install/dylib_fix.py -p "$<TARGET_FILE:${PROJECT_NAME}>" -o "${CMAKE_SOURCE_DIR}/support" -l "${torch_dir}/libtorch" "${CMAKE_BINARY_DIR}/_deps" "${CMAKE_SOURCE_DIR}/../env" "${HOMEBREW_PREFIX}" --sign_id "${SIGN_ID}"
				COMMENT "Fixing libraries, certificates, permissions, codesigning, quarantine"
				VERBATIM
			)
		else()
			add_custom_command(
				TARGET ${PROJECT_NAME}
				POST_BUILD 
				COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../install/dylib_fix.py -p "$<TARGET_FILE:${PROJECT_NAME}>" -o "${CMAKE_SOURCE_DIR}/support" -l "${torch_dir}/libtorch" "${CMAKE_BINARY_DIR}/_deps" "${CMAKE_SOURCE_DIR}/../env" "${HOMEBREW_PREFIX}" --sign_id "${SIGN_ID}"
				COMMENT "Fixing libraries, permissions, codesigning, quarantine"
				VERBATIM
			)
		endif()
	else()
		add_custom_command(
			TARGET ${PROJECT_NAME}
			POST_BUILD 
			COMMAND codesign --force --deep --sign "${SIGN_ID}" "$<TARGET_FILE_DIR:${PROJECT_NAME}>/../.."
			COMMENT "Codesigning ${PROJECT_NAME}"
			VERBATIM
		)
	endif()
endif()
